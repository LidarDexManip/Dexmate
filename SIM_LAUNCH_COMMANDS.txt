# --- Pico hand tracking (arm) + Wuji glove (wrist and fingers) ---

scripts/launch_wuji_vega_pico_hand.sh

#   .venv/bin/python -m wuji_teleop.vega_arm_sim \

# Terminal 1 (leave running):

  --host 192.168.1.65 --port 8090 --mode wireless --input hand \
  --disable-world-frame-snap --no-shm
# Terminal 2 (restart as often as needed):
.venv/bin/python -m wuji_teleop.vega_arm_sim \
  --arm-input pico-hand --finger-input glove \
  --glove-sn WG1JA06260812032 \
  --report /tmp/vega-pico-hand-live.json
