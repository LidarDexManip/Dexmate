# Vega-1U + Wuji Hand 2 — simulation launch commands
# Run from the repo root. New machine: see NEW_MACHINE.md.
#
# Every scripts/launch_*.sh appends what it ran to SIM_RUN_LOG.txt (tail SIM_RUN_LOG.txt).
# Launchers auto-detect DexStream, the Wi-Fi IP and the glove. Overrides:
#   GLOVE_SN=<serial>  WIFI_INTERFACE=<iface>  DEXSTREAM_BIN=<path>  PICO_SCALE=<x>
# Reports and CSV logs go to /tmp/.

# --- One-time setup (installs everything into .venv, including DexStream) ---
scripts/setup_sim.sh

# --- Glove network (glove 192.168.1.100, laptop 192.168.1.50; temporary profile) ---
bash scripts/configure_wuji_network.sh
.venv/bin/python -m wuji_teleop discover --seconds 3

# --- Joint limit tool (no glove, no headset): jog window + MuJoCo viewer ---
scripts/launch_vega_joint_limits.sh
#   = .venv/bin/python -m wuji_teleop.vega_arm_range

# --- Hand only: glove fingers, arm fixed ---
scripts/launch_wuji_vega_hand_only.sh [SERIAL]

# --- Glove fingers + wrist from palm IMU (shoulder/elbow fixed) ---
scripts/launch_wuji_vega_glove_wrist.sh [SERIAL]

# --- Glove fingers + whole arm from palm IMU tilt (no Pico) ---
scripts/launch_wuji_vega_glove_arm.sh [SERIAL]

# --- Hand + Pico: headset hand tracking -> arm j1-j4; glove -> wrist j5-j7 + fingers ---
scripts/launch_wuji_vega_pico_hand.sh [WIFI_IP] [SERIAL]
# Default translation scale is 0.6; override only after checking reach and jitter.
PICO_SCALE=1.3 scripts/launch_wuji_vega_pico_hand.sh
#   = .venv/bin/dexstream --host <wifi-ip> --port 8090 --mode wireless --input hand \
#       --disable-world-frame-snap --no-shm
#   + .venv/bin/python -m wuji_teleop.vega_arm_sim --arm-input pico-hand \
#       --finger-input glove [--glove-sn SERIAL] --report /tmp/vega-pico-hand-live.json \
#       [--pico-scale X] [--log-csv PATH]
# Keep the Pico page up while restarting the sim:
#   Terminal 1: scripts/launch_dexstream.sh hand        (leave running)
#   Terminal 2: scripts/launch_wuji_vega_pico_hand.sh    (sees DexStream, starts only the sim)

# --- Bilateral hand + Pico: both tracked hands -> both arms; both gloves -> wrists/fingers ---
scripts/launch_wuji_vega_bimanual_pico.sh [WIFI_IP] [LEFT_SERIAL] [RIGHT_SERIAL]
# Auto-selects exactly one discovered left glove and one right glove when serials are omitted.
# Press R in MuJoCo to anchor both sides; logs to /tmp/vega-bimanual-pico-<time>.csv.

# --- Pico controller rotates the wrist only (glove fingers) ---
scripts/launch_wuji_vega_pico_wrist.sh [WIFI_IP] [SERIAL]
#   or: scripts/launch_dexstream.sh paddle  in a separate terminal

# --- Manual arm pose from the command line ---
scripts/launch_wuji_vega_manual_arm.sh [J1 J2 J3 J4 J5 J6 J7]

# --- Legacy: full 7-joint arm from the Pico controller (needs scripts/launch_dexstream.sh paddle) ---
.venv/bin/python -m wuji_teleop.vega_arm_sim --arm-input pico-controller \
  --finger-input glove --report /tmp/vega-pico-controller-live.json

# --- Headless checks (no window) ---
.venv/bin/python -m wuji_teleop.vega_arm_sim --arm-input synthetic --finger-input synthetic \
  --headless --duration 5
.venv/bin/python -m wuji_teleop.vega_arm_sim --arm-input glove-arm --finger-input glove \
  --headless --duration 10 --report /tmp/vega-arm-check.json

# --- Unit tests ---
.venv/bin/python -m unittest tests.test_vega_arm_sim tests.test_vega_pico_hand \
  tests.test_vega_pico_wrist tests.test_vega_bimanual_sim \
  tests.test_imu_arm_source tests.test_vr_pose_source
